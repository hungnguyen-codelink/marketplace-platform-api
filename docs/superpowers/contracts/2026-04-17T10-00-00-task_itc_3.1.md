task_id: 3.1
task_name: "Products Backend (Phase 3)"

implementation_scope:
  components:
    - "ProductsService with CRUD and search methods (backend/src/modules/products/service.ts)"
    - "ProductsController with 5 handler methods (backend/src/modules/products/controller.ts)"
    - "ProductsRouter with Zod schemas and middleware stack (backend/src/modules/products/router.ts)"
    - "App.ts modification to register /api/products route (backend/src/app.ts)"
    - "Unit tests for ProductsService (backend/tests/unit/modules/products/service.test.ts)"
    - "Integration tests for Products endpoints (backend/tests/integration/modules/products/endpoints.integration.test.ts)"

  integration_surfaces:
    - "PostgreSQL products table (products.id, shop_id, fakestore_id, title, description, price, image_url, category, stock, aggregate_rating, review_count, created_at, updated_at)"
    - "PostgreSQL shops table via FK shop_id and seller verification"
    - "Users table (existing, referenced through shops.seller_id)"
    - "Sessions table (existing, used by authenticate middleware)"
    - "Express middleware: authenticate, requireRole, validate, asyncHandler"
    - "JWT token validation through authenticate middleware"
    - "DB constraint enforcement: price > 0, stock >= 0 AND stock <= 999999"
    - "Numeric field enforcement: price NUMERIC(10,2), stock INTEGER"

test_contract:
  tiers_required: [unit, integration]
  rationale: "Unit tests verify service layer DB query correctness, search logic, validation, and error mapping; integration tests verify middleware stack, auth/role enforcement, field validation, ownership checks, pagination, and HTTP response contracts."

  unit:
    commands:
      - "npm test -- tests/unit/modules/products/service.test.ts"
    must_cover:
      - "createProduct() accepts valid title, description, price, image_url, category, stock; inserts to DB with seller's shop_id; returns ProductResponse without fakestore_id"
      - "createProduct() throws ValidationError if price <= 0"
      - "createProduct() throws ValidationError if price is not numeric with exactly 2 decimal places (e.g., 10.999 invalid, 10.99 valid)"
      - "createProduct() throws ValidationError if stock < 0"
      - "createProduct() throws ValidationError if stock > 999999"
      - "createProduct() throws ValidationError if stock is not integer (e.g., 5.5 invalid)"
      - "createProduct() throws NotFoundError if seller has no shop"
      - "createProduct() returns ProductResponse with fields: id, shop_id, title, description, price, image_url, category, stock, aggregate_rating, review_count, created_at, updated_at (does NOT include fakestore_id)"
      - "createProduct() price field in response is numeric with 2 decimal places"
      - "createProduct() stock field in response is integer with no decimals"
      - "getProductById() returns ProductResponse for any product (public); excludes fakestore_id; includes all required fields with correct types"
      - "getProductById() throws NotFoundError if product does not exist"
      - "getProducts() returns paginated list with default page 1, limit 10"
      - "getProducts() respects pagination query params (page, limit)"
      - "getProducts() excludes fakestore_id from all product objects in response"
      - "getProducts() filters by category with exact match on products.category field"
      - "getProducts() searches keyword (case-insensitive) matching title, description, or category fields"
      - "getProducts() returns empty array for non-matching search keyword"
      - "getProducts() returns empty array for non-matching category"
      - "getProducts() returns correct subset when both search and category filters applied simultaneously"
      - "updateProduct() accepts partial payload (any of title, description, price, image_url, category, stock)"
      - "updateProduct() updates only provided fields; leaves omitted fields unchanged"
      - "updateProduct() returns updated ProductResponse without fakestore_id"
      - "updateProduct() throws NotFoundError if product does not exist"
      - "updateProduct() throws ForbiddenError if product.shop_id does not match seller.shop_id"
      - "updateProduct() revalidates price > 0 and price numeric with 2 decimals on update"
      - "updateProduct() revalidates stock bounds (>= 0, <= 999999) and integer constraint on update"
      - "updateProduct() updates updated_at timestamp to current time"
      - "deleteProduct() hard-deletes from DB; returns no body"
      - "deleteProduct() throws NotFoundError if product does not exist or already deleted"
      - "deleteProduct() throws ForbiddenError if product.shop_id does not match seller.shop_id"
      - "All service method errors map to correct AppError subclasses (ValidationError 400, AuthError 401, ForbiddenError 403, NotFoundError 404)"

  integration:
    commands:
      - "npm test -- tests/integration/modules/products/endpoints.integration.test.ts"
    must_cover:
      - "POST /api/products with valid payload (title, description, price, image_url, category, stock), seller auth, returns 201 with product object (no fakestore_id)"
      - "POST /api/products with missing auth returns 401 Unauthorized"
      - "POST /api/products with buyer role returns 403 Forbidden"
      - "POST /api/products with seller role but no shop returns 403 Forbidden or 404 Not Found"
      - "POST /api/products with price field missing returns 400 Bad Request with field error"
      - "POST /api/products with price <= 0 returns 400 Bad Request with field error"
      - "POST /api/products with price not numeric (non-numeric string) returns 400 Bad Request"
      - "POST /api/products with price more than 2 decimal places (e.g., 10.999) returns 400 Bad Request"
      - "POST /api/products with stock field missing returns 400 Bad Request with field error"
      - "POST /api/products with stock < 0 returns 400 Bad Request with field error"
      - "POST /api/products with stock > 999999 returns 400 Bad Request with field error"
      - "POST /api/products with stock as decimal (e.g., 5.5) returns 400 Bad Request"
      - "POST /api/products response includes created_at and updated_at timestamps"
      - "GET /api/products (public) returns paginated list with default 10 items"
      - "GET /api/products returns all product fields: id, shop_id, title, description, price, image_url, category, stock, aggregate_rating, review_count, created_at, updated_at"
      - "GET /api/products response does NOT include fakestore_id in any product object"
      - "GET /api/products?search=keyword returns products matching keyword case-insensitively in title, description, or category"
      - "GET /api/products?search=UPPERCASE returns same results as ?search=lowercase (case-insensitive)"
      - "GET /api/products?search=keyword with no matches returns empty array"
      - "GET /api/products?category=electronics returns only products with category='electronics' (exact match)"
      - "GET /api/products?category=nonexistent returns empty array"
      - "GET /api/products?page=2&limit=5 respects pagination; returns items from correct offset with correct count"
      - "GET /api/products?page=1&limit=1 returns exactly 1 item"
      - "GET /api/products?search=keyword&category=electronics combines both filters (AND logic)"
      - "GET /api/products/:id (public) returns single product detail; no fakestore_id; includes all fields"
      - "GET /api/products/:id with non-existent id returns 404 Not Found"
      - "PUT /api/products/:id with seller auth, ownership verified, partial payload returns 200 with updated product"
      - "PUT /api/products/:id updating price validates price > 0 and 2 decimal places; returns 400 if invalid"
      - "PUT /api/products/:id updating stock validates bounds and integer constraint; returns 400 if invalid"
      - "PUT /api/products/:id updates updated_at to current time"
      - "PUT /api/products/:id without auth returns 401 Unauthorized"
      - "PUT /api/products/:id with buyer role returns 403 Forbidden"
      - "PUT /api/products/:id with seller not owning product (different shop_id) returns 403 Forbidden"
      - "PUT /api/products/:id with non-existent product returns 404 Not Found"
      - "DELETE /api/products/:id with seller auth, ownership verified returns 204 No Content"
      - "DELETE /api/products/:id without auth returns 401 Unauthorized"
      - "DELETE /api/products/:id with buyer role returns 403 Forbidden"
      - "DELETE /api/products/:id with seller not owning product (different shop_id) returns 403 Forbidden"
      - "DELETE /api/products/:id with non-existent product returns 404 Not Found"
      - "DELETE /api/products/:id on deleted product returns 404 Not Found (idempotence)"
      - "Responses must NOT include fakestore_id in any endpoint"
      - "Price field in responses is numeric type with exactly 2 decimal places (from NUMERIC(10,2) DB column)"
      - "Stock field in responses is integer type with no decimal component"
      - "All timestamps (created_at, updated_at) are ISO 8601 formatted strings"

    required_services:
      - "PostgreSQL test database (via DATABASE_URL env var, typically postgresql://localhost:5432/test)"
      - "Environment variables: DATABASE_URL, JWT_SECRET, SESSION_TTL_SECONDS"
      - "Jest test runner with ts-jest preset (backend/jest.config.ts with maxWorkers: 1)"
