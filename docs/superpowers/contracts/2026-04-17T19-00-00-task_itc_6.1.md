# Task ITC 6.1 — Backend Reviews API

```yaml
task_id: 6.1
task_name: "Backend Reviews API"

implementation_scope:
  components:
    - "Reviews service class (/backend/src/modules/reviews/service.ts)"
    - "Reviews controller class (/backend/src/modules/reviews/controller.ts)"
    - "Reviews router with Zod validation (/backend/src/modules/reviews/router.ts)"
    - "App registration of reviews router (/backend/src/app.ts)"
    - "Unit tests for reviews service (/backend/tests/unit/modules/reviews/service.test.ts)"
    - "Unit tests for reviews controller and router (/backend/tests/unit/modules/reviews/controller.test.ts)"
    - "Integration tests (/backend/tests/integration/modules/reviews/endpoints.integration.test.ts)"

  integration_surfaces:
    - "PostgreSQL database: orders table (status field check)"
    - "PostgreSQL database: order_items table (lookup, UNIQUE constraint on reviews.order_item_id)"
    - "PostgreSQL database: reviews table (INSERT, atomicity)"
    - "PostgreSQL database: products table (aggregate_rating and review_count UPDATE)"

test_contract:
  tiers_required: [unit, integration]
  rationale: "Unit tests cover business logic and error handling at the service/controller level. Integration tests verify atomic rating recalculation and UNIQUE constraint enforcement against live database."

  unit:
    commands:
      - "cd /backend && npm test -- --testPathPattern=reviews/service"
      - "cd /backend && npm test -- --testPathPattern=reviews/controller"
    must_cover:
      - "AC-04: Service method rejects review submission with ForbiddenError when parent order status != 'completed'"
      - "AC-04: Controller returns HTTP 403 when ForbiddenError thrown"
      - "REV-05: Service method rejects second review on same order_item with ConflictError"
      - "REV-05: Controller returns HTTP 409 Conflict when ConflictError thrown"
      - "Service rejects rating=0 with ValidationError"
      - "Service rejects rating=6 with ValidationError"
      - "Service rejects rating=3.5 (non-integer float) with ValidationError"
      - "Service throws NotFoundError when order_item_id does not exist"
      - "Service throws NotFoundError when parent order does not exist"
      - "Service throws NotFoundError when product_id is not found"
      - "Service validates buyer_id matches order buyer_id (ownership check)"
      - "Controller maps all error types to correct HTTP status codes (403/409/400/404)"
      - "Router includes POST /api/reviews with authenticate and requireRole('buyer') middleware"
      - "Router Zod validation for order_item_id (UUID), rating (1-5 integer), text (optional string)"
      - "Success response returns 201 with review object including text (or null)"

  integration:
    commands:
      - "cd /backend && npm test -- --testPathPattern=reviews/endpoints.integration"
    must_cover:
      - "AC-04: POST /api/reviews returns 403 when order.status = 'pending'"
      - "AC-04: POST /api/reviews returns 403 when order.status in ['confirmed', 'shipped', 'delivered']"
      - "AC-04: POST /api/reviews returns 201 when order.status = 'completed'"
      - "AC-04: Unauthenticated request returns 401"
      - "REV-05: First review on order_item succeeds (201)"
      - "REV-05: Second review on same order_item fails (409 Conflict)"
      - "REV-05: UNIQUE constraint violation rolls back — no partial write"
      - "AC-10: After 0 reviews, product.aggregate_rating = 0.00 and review_count = 0 (initial state)"
      - "AC-10: After 1 review with rating=5, product.aggregate_rating = 5.00 and review_count = 1"
      - "AC-10: After 2 reviews (ratings 4 and 5), product.aggregate_rating = 4.50 and review_count = 2"
      - "AC-10: After 3 reviews (ratings 3, 4, 5), product.aggregate_rating = 4.00 and review_count = 3"
      - "AC-10: Rating recalculation with 5 reviews (ratings 1,2,3,4,5), aggregate_rating = 3.00 and review_count = 5"
      - "Integration: Buyer cannot review items from orders they don't own"
      - "Integration: Buyer cannot bypass order status check"
      - "Integration: Two concurrent review submissions on same order_item — one succeeds, one gets 409"
    required_services:
      - "PostgreSQL running and accessible via env var DATABASE_URL"
      - "Migrations 001-008 applied to test database"
      - "Environment variables: DATABASE_URL, JWT_SECRET, SESSION_TTL_SECONDS=3600"
      - "Test database isolated (each test cleans up state)"

acceptance_criteria:
  unit: all pass
  integration: all pass
  no_regression: true

sign_off:
  coding_agent: ✅
  testing_agent: ✅
```
