# ITC: Task 1.1 — Auth Backend (Phase 1) - Round 4

```yaml
task_id: "1.1"
task_name: "Auth Backend (Phase 1)"

implementation_scope:
  components:
    - "POST /api/auth/register endpoint — bcrypt hash, users insert, 201 with token, 409 on dupe email"
    - "POST /api/auth/login endpoint — bcrypt verify, JWT issuance, sessions insert, 200 with token, 401 on bad creds"
    - "POST /api/auth/logout endpoint — require auth, delete session row, 204"
    - "authenticate middleware — JWT verify + sessions revocation check, attach req.user"
    - "requireRole middleware — role guard, 403 on mismatch"
    - "validate middleware — Zod, 400 with field-level errors"

  integration_surfaces:
    - "PostgreSQL database — users, sessions, and role schema"
    - "JWT token issuance (HS256) and verification"
    - "Bcrypt password hashing (salt factor: 10)"
    - "Redis-based session revocation tracking (optional, covered by sessions table)"
    - "Request body validation via Zod"

test_contract:
  tiers_required: [unit, integration]
  rationale: "Unit tests verify middleware behavior and payload shape in isolation; integration tests verify end-to-end flows, concurrent edge cases, and database constraints."

  unit:
    commands:
      - "npm test -w backend -- --testPathPattern=\"tests/unit/modules/auth/service\" && npm test -w backend -- --testPathPattern=\"tests/unit/middleware/auth\""

    must_cover:
      - "validate middleware returns 400 with structured field errors (one error per invalid field) for validation failures"
      - "validate middleware passes request with validated data attached when validation succeeds"
      - "JWT payload contains exactly { id, email, role } with no extra fields"
      - "JWT signature verification fails for tampered tokens (returns 401)"
      - "authenticate middleware returns 401 on missing Authorization header"
      - "authenticate middleware returns 401 on malformed Authorization header (not 'Bearer <token>')"
      - "authenticate middleware returns 401 on expired JWT (expires_at <= Date.now())"
      - "authenticate middleware attaches decoded user object to req.user on success"
      - "authenticate middleware returns 401 on valid JWT but revoked session (token_hash not in sessions table)"
      - "requireRole middleware returns 403 when user.role does not match required role"
      - "requireRole middleware passes through when user.role matches"
      - "Bcrypt hash of password differs on each invocation (salt randomness verified)"
      - "Bcrypt verify returns false for plaintext compared to hash (one-way hashing confirmed)"

  integration:
    commands:
      - "npm test -w backend -- --testPathPattern=\"tests/integration/modules/auth\""

    must_cover:
      - "POST /api/auth/register with valid email/password/full_name/role returns 201 with user object and token"
      - "POST /api/auth/register with missing role field returns 400 with validation error"
      - "POST /api/auth/register with role='admin' returns 400 validation error (invalid enum value, not 403)"
      - "POST /api/auth/register with role='invalid' returns 400 validation error"
      - "POST /api/auth/register with role='buyer' succeeds and creates user with role='buyer'"
      - "POST /api/auth/register with role='seller' succeeds and creates user with role='seller'"
      - "POST /api/auth/register with duplicate email in concurrent requests: second request returns 409"
      - "POST /api/auth/register with email containing whitespace is handled consistently (trimmed or rejected with 400)"
      - "POST /api/auth/register creates sessions row with expires_at = now + SESSION_TTL_SECONDS"
      - "POST /api/auth/register password is hashed with bcrypt before insertion (verified by comparing hash to plaintext fails)"
      - "POST /api/auth/login with valid email/password returns 200 with user object and token"
      - "POST /api/auth/login with invalid email returns 401 'Invalid credentials'"
      - "POST /api/auth/login with invalid password returns 401 'Invalid credentials'"
      - "POST /api/auth/login creates sessions row linked to user and token_hash"
      - "POST /api/auth/logout requires Authorization header (401 if missing)"
      - "POST /api/auth/logout with malformed Authorization header returns 401"
      - "POST /api/auth/logout with valid JWT deletes corresponding sessions row, returns 204"
      - "POST /api/auth/logout with valid JWT from revoked session returns 401"
      - "POST /api/auth/logout with User A's token correctly deletes User A's session, not User B's"
      - "Session expiry logic: token with expires_at in past is rejected as revoked (returns 401)"
      - "Validation error responses include structured field errors, not generic messages"
      - "Validation failure on register (missing email) returns 400 with { message, errors: { email: ['required'] } }"
      - "Validation failure on login (invalid email format) returns 400 with structured field errors"

    required_services:
      - "PostgreSQL database (test DB at DATABASE_URL)"
      - "beforeAll drops and recreates tables (users, sessions, roles)"
      - "afterEach cleans up test data (truncates test user/session records)"
      - "afterAll cleans up all tables and closes DB connection"
      - "Environment variables: JWT_SECRET (non-empty string), SESSION_TTL_SECONDS=3600, DATABASE_URL=postgres://test-db, PORT=3001 or ephemeral"
      - "Date.now() is mockable in tests (Jest mocking for token expiry scenarios)"

acceptance_criteria:
  unit: all pass
  integration: all pass
  no_regression: true

response_format_contract:
  register_success: "201 Created with { user: { id: UUID, email: string, full_name: string, role: 'buyer'|'seller' }, token: string }"
  login_success: "200 OK with { user: { id: UUID, email: string, full_name: string, role: 'buyer'|'seller' }, token: string }"
  logout_success: "204 No Content (empty body)"
  error_400: "{ message: string, errors?: { [field: string]: string[] } } — validation failures MUST include errors object; other 400s may omit it"
  error_401: "{ message: string } — 'Invalid credentials', 'Unauthorized', or 'Token revoked'"
  error_403: "{ message: 'Forbidden' }"
  error_409: "{ message: 'Email already exists' }"

clarifications:
  role_validation: "The role field is REQUIRED in POST /api/auth/register. Acceptable values are 'buyer' or 'seller'. Values outside this enum (including 'admin', 'null', or missing) are validation errors returning 400, not 403. There is NO default role; all validation failures return 400."
  admin_self_register: "Attempting to register with role='admin' is a validation error (400), not a Forbidden response (403). The enum constraint at the input validation layer prevents invalid role values; this is not a permission issue but an input validity issue."
  logout_session_isolation: "POST /api/auth/logout deletes the session row associated with the provided token. A valid JWT from User A will only delete User A's session, not User B's, because sessions are keyed by token_hash. This test covers session isolation correctness."
  email_whitespace: "email field must be trimmed before database insertion OR validation must reject untrimmed input. Test must verify consistent behavior (either always trim or always reject)."
  token_expiry_testing: "JWT expiry is testable by mocking Date.now() and setting expires_at to a past timestamp. This enables deterministic testing of the authenticate middleware's expiry check."

amendments_from_round_3:

  - "Issue 3 — test command format: Replaced testNamePattern with testPathPattern for more reliable test routing. Replaced nonexistent test:integration script with test script. Unit tests now run: 'npm test -w backend -- --testPathPattern=\"tests/unit/modules/auth/service\" && npm test -w backend -- --testPathPattern=\"tests/unit/middleware/auth\"'. Integration tests now run: 'npm test -w backend -- --testPathPattern=\"tests/integration/modules/auth\"'."
```
