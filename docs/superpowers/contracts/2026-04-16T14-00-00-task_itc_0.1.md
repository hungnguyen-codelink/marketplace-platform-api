# ITC: Task 0.1 — Monorepo + Docker Setup

```yaml
task_id: "0.1"
task_name: "Monorepo + Docker Setup"

implementation_scope:
  components:
    - "Root package.json with npm workspaces definition (package.json)"
    - "Docker Compose services (postgres:16-alpine, redis:7-alpine) (docker-compose.yml)"
    - "Environment variables example file (.env.example)"
    - "Backend package.json with Express/TypeScript/Jest dependencies (backend/package.json)"
    - "Backend TypeScript configuration (backend/tsconfig.json)"
    - "Backend Jest test configuration (backend/jest.config.ts)"
    - "Frontend package.json with React/TypeScript/Tailwind dependencies (frontend/package.json)"
    - "Frontend TypeScript configuration (frontend/tsconfig.json)"
    - "Frontend Tailwind CSS configuration (frontend/tailwind.config.ts)"

  integration_surfaces:
    - "Docker daemon — instantiating and running containerized services"
    - "PostgreSQL database on port 5432 — connectivity"
    - "Redis on port 6379 — connectivity and command execution"
    - "npm workspaces — coordinating package resolution across backend/frontend/e2e"

test_contract:
  tiers_required: [unit, integration]
  rationale: "Unit tests verify configuration file structure and content; integration tests verify Docker containers start and services are reachable."

  unit:
    commands:
      - "node -e \"const pkg = require('./package.json'); console.assert(pkg.workspaces && pkg.workspaces.length === 3); console.log('✓ root package.json workspaces')\""
      - "node -e \"const pkg = require('./backend/package.json'); console.assert(pkg.devDependencies.typescript && pkg.devDependencies.jest); console.log('✓ backend deps')\""
      - "node -e \"const pkg = require('./frontend/package.json'); console.assert(pkg.dependencies.react && pkg.dependencies.tailwindcss); console.log('✓ frontend deps')\""
      - "node -e \"const fs = require('fs'); const backend = JSON.parse(fs.readFileSync('./backend/tsconfig.json', 'utf8')); console.assert(backend.compilerOptions && backend.compilerOptions.strict); console.log('✓ backend tsconfig valid')\""
      - "node -e \"const fs = require('fs'); const frontend = JSON.parse(fs.readFileSync('./frontend/tsconfig.json', 'utf8')); console.assert(frontend.compilerOptions && frontend.compilerOptions.jsx); console.log('✓ frontend tsconfig valid')\""
      - "node -e \"const fs = require('fs'); const jestCfg = fs.readFileSync('./backend/jest.config.ts', 'utf8'); console.assert(jestCfg.includes('ts-jest') && jestCfg.includes('testMatch')); console.log('✓ jest.config.ts valid')\""
      - "node -e \"const fs = require('fs'); const twCfg = fs.readFileSync('./frontend/tailwind.config.ts', 'utf8'); console.assert(twCfg.includes('content') && twCfg.includes('src/')); console.log('✓ tailwind.config.ts valid')\""
      - "node -e \"const fs = require('fs'); const dc = fs.readFileSync('./docker-compose.yml', 'utf8'); console.assert(dc.includes('postgres:16-alpine') && dc.includes('redis:7-alpine')); console.log('✓ docker-compose.yml images valid')\""
      - "node -e \"const fs = require('fs'); const env = fs.readFileSync('./.env.example', 'utf8'); ['DATABASE_URL','REDIS_URL','JWT_SECRET','SESSION_TTL_SECONDS','RATE_LIMIT_WINDOW_MS','FAKESTORE_BASE_URL'].forEach(v => console.assert(env.includes(v), v + ' missing')); console.log('✓ .env.example complete')\""

    must_cover:
      - "Root package.json declares workspaces: [backend, frontend, e2e]"
      - "Backend package.json includes: express, pg, ioredis, bcryptjs, jsonwebtoken, zod, typescript, jest, ts-node, ts-jest"
      - "Frontend package.json includes: react, react-dom, react-router-dom, axios, @tanstack/react-query, tailwindcss"
      - "Backend tsconfig.json targets ES2020, enables strict mode, resolves from src/"
      - "Frontend tsconfig.json targets ES2020, enables JSX, resolves from src/"
      - "Backend jest.config.ts references ts-jest and testMatch"
      - "Frontend tailwind.config.ts content path covers src/**"
      - "docker-compose.yml defines postgres:16-alpine and redis:7-alpine"
      - ".env.example includes all required env vars (DATABASE_URL, REDIS_URL, JWT_SECRET, SESSION_TTL_SECONDS, RATE_LIMIT_WINDOW_MS, FAKESTORE_BASE_URL)"

  integration:
    commands:
      - "cp .env.example .env && docker-compose config > /dev/null && echo '✓ docker-compose.yml loads'"
      - "docker-compose up -d && sleep 3 && docker-compose ps | grep -E '(postgres|redis)' | grep 'Up' && docker-compose down -v && echo '✓ services running'"
      - "docker-compose up -d && sleep 3 && docker-compose exec -T postgres pg_isready -U marketplace -d marketplace && docker-compose down -v && echo '✓ postgres ready'"
      - "docker-compose up -d && sleep 3 && docker-compose exec -T redis redis-cli ping && docker-compose down -v && echo '✓ redis pong'"

    must_cover:
      - "docker-compose.yml loads successfully with environment variables from .env"
      - "docker-compose up -d completes and both postgres and redis services reach running state"
      - "postgres service accepts connections (pg_isready returns 0)"
      - "redis service responds with PONG"
      - "docker-compose down -v cleanly stops and removes containers"

    required_services:
      - "Docker daemon running"
      - "npm >= 7.x"
      - "node >= 16.x"
      - ".env file (created from .env.example)"

acceptance_criteria:
  unit: all pass
  integration: all pass
  no_regression: true

sign_off:
  coding_agent: ✅
  testing_agent: ✅
```
