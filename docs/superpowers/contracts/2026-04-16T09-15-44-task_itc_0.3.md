# ITC: Task 0.3 — DB Migration Runner & Schema

```yaml
task_id: "0.3"
task_name: "DB Migration Runner & Schema"

implementation_scope:
  components:
    - "Migration runner script (backend/src/db/migrations/runner.ts)"
    - "SQL migration files 001-008 (backend/src/db/migrations/*.sql)"
    - "Migration utilities module (backend/src/db/migrationUtils.ts)"
    - "Updated backend/package.json with db:migrate script"

  integration_surfaces:
    - "PostgreSQL database connection via DATABASE_URL environment variable"
    - "File system - reads SQL files from migrations directory at runtime"

test_contract:
  tiers_required: [unit, integration]
  rationale: "Unit tests verify migration runner logic in isolation; integration tests verify schema creation against a real PostgreSQL database with constraint enforcement."

  unit:
    commands:
      - "npm run test -w backend -- --testPathPattern='db/migrations/runner.test'"
      - "npm run test -w backend -- --testPathPattern='db/migrationUtils.test'"
    must_cover:
      - "Migration runner loads SQL files from migrations directory in lexicographic order"
      - "Migration runner skips already-applied migrations (idempotency)"
      - "Migration runner handles missing DATABASE_URL environment variable with clear error"
      - "migrationUtils.hashFile() produces consistent hashes"
      - "migrationUtils.readMigrationFiles() returns files sorted by name"
      - "Error handling when SQL file is malformed or unreadable"

  integration:
    commands:
      - "npm run test -w backend -- --testPathPattern='db/migrations.integration.test'"
    must_cover:
      - "_migrations table is created on first run with columns: filename text NOT NULL, hash text NOT NULL, applied_at timestamptz DEFAULT now()"
      - "Each migration file is executed in lexicographic order"
      - "users table: id uuid PK, email text UNIQUE NOT NULL, password_hash, full_name, role CHECK IN ('buyer','seller','admin'), email_verified DEFAULT false, terms_accepted DEFAULT false, created_at"
      - "shops table: UNIQUE seller_id FK→users"
      - "products table: price>0, stock 0-999999, fakestore_id UNIQUE nullable"
      - "orders table: status DEFAULT 'pending' CHECK IN ('pending','confirmed','shipped','delivered','completed')"
      - "order_items: quantity CHECK >0"
      - "reviews: rating CHECK 1-5, order_item_id UNIQUE"
      - "categories: name UNIQUE, id SERIAL"
      - "sessions: token_hash PK, expires_at timestamptz"
      - "All foreign key constraints enforced"
      - "Running migrations twice does not error (idempotency)"
      - "Constraint violations rejected: invalid role, price<0, stock>999999, rating outside 1-5, quantity<=0"
      - "UNIQUE constraints enforcement (email, seller_id, fakestore_id, order_item_id)"
      - "Column DEFAULT values enforced (email_verified=false, terms_accepted=false, stock=0, status='pending')"
      - "Numeric constraints: products.price > 0, stock 0-999999, reviews.rating 1-5, order_items.quantity > 0"
    required_services:
      - "PostgreSQL 16+ database with fresh schema (via docker-compose postgres service or TEST_DATABASE_URL)"

acceptance_criteria:
  unit: all pass
  integration: all pass
  no_regression: true

sign_off:
  coding_agent: ✅
  testing_agent: ✅
```
