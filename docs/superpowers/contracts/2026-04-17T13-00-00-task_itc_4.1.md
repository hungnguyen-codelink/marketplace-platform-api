# ITC: Task 4.1 — FakeStore Integration Backend

```yaml
task_id: 4.1
task_name: "FakeStore Integration Backend (Phase 4)"

implementation_scope:
  components:
    - "FakeStore HTTP client with retry/timeout logic (backend/src/modules/fakestore/client.ts)"
    - "FakeStore service: product fetching, category caching, product import with upsert logic (backend/src/modules/fakestore/service.ts)"
    - "FakeStore controller: request/response handling for 3 endpoints (backend/src/modules/fakestore/controller.ts)"
    - "FakeStore router: 3 endpoints with auth/validation (backend/src/modules/fakestore/router.ts)"
    - "Zod schemas for import request validation (backend/src/modules/fakestore/schemas.ts)"
    - "Update app.ts to register fakestore router at /api/fakestore (backend/src/app.ts)"
    - "Unit tests for FakeStore client (backend/tests/unit/modules/fakestore/client.test.ts)"
    - "Unit tests for FakeStore service (backend/tests/unit/modules/fakestore/service.test.ts)"
    - "Integration tests for FakeStore endpoints (backend/tests/integration/modules/fakestore/endpoints.integration.test.ts)"

  integration_surfaces:
    - "FakeStore API at https://fakestoreapi.com (GET /products, GET /categories)"
    - "Redis client for category caching via existing redisGet/redisSet/redisDel helpers (fail-silent)"
    - "PostgreSQL: products table (upsert by fakestore_id), categories table (fallback), shops table (seller ownership)"
    - "authenticate + requireRole('seller') middleware"

test_contract:
  tiers_required: [unit, integration]
  rationale: "Unit tests cover HTTP client retry/timeout, field defaults/mapping, and upsert decision logic in isolation. Integration tests verify end-to-end flows including Redis→DB cache fallback, auth enforcement, and product import with overwrite semantics."

  unit:
    commands:
      - "npm test -- tests/unit/modules/fakestore/client.test.ts"
      - "npm test -- tests/unit/modules/fakestore/service.test.ts"
    must_cover:
      # HTTP client
      - "HTTP client: request aborts after FAKESTORE_TIMEOUT_MS (5000ms)"
      - "HTTP client: 5xx responses (500, 502, 503, 504) trigger retry"
      - "HTTP client: network errors (ECONNREFUSED, ETIMEDOUT) trigger retry"
      - "HTTP client: exponential backoff — 500ms, 1000ms, 2000ms for FAKESTORE_RETRY_ATTEMPTS=3"
      - "HTTP client: returns descriptive error after exhausting all retries (no system crash)"
      - "HTTP client: successful response returns parsed JSON"
      # Field mapping and defaults
      - "Field mapping: FakeStore id → fakestore_id in products table"
      - "Field mapping: FakeStore rating.rate → aggregate_rating; missing/null defaults to 0"
      - "Field mapping: FakeStore rating.count → review_count; missing/null defaults to 0"
      - "Field defaults: title defaults to 'Untitled Product' when missing/null"
      - "Field defaults: description defaults to '' when missing/null"
      - "Field defaults: price defaults to 0.01 when missing/null/non-numeric/NaN"
      - "Field defaults: image_url defaults to null when missing/null"
      - "Field defaults: category defaults to 'uncategorized' when missing/null"
      - "Field defaults: stock defaults to 0 (FakeStore has no stock field)"
      # Upsert logic
      - "Upsert: overwrite=false preserves existing price, description, stock; updates title, image_url, category, aggregate_rating, review_count"
      - "Upsert: overwrite=true replaces all fields including price, description, stock with FakeStore values"
      # Category cache
      - "Category cache: reads from Redis key 'categories:fakestore' with FAKESTORE_CATEGORY_CACHE_TTL_SECONDS TTL"
      - "Category cache: on cache hit, returns cached array without fetching from FakeStore"
      - "Category cache: on Redis unavailability, falls back to DB categories table"
      - "Category cache: on DB table empty, fetches from FakeStore, populates DB, caches in Redis, returns result"
      - "Category cache: returns string array of category names"
      # Validation
      - "Validation: ids must be non-empty array of integers"
      - "Validation: overwrite must be boolean (optional, defaults to false)"

  integration:
    commands:
      - "npm test -- tests/integration/modules/fakestore/endpoints.integration.test.ts"
    must_cover:
      # GET /api/fakestore/products
      - "GET /api/fakestore/products: returns 401 without auth token"
      - "GET /api/fakestore/products: returns 403 for buyer/admin role"
      - "GET /api/fakestore/products: seller with valid auth returns all FakeStore products as array with field mappings applied"
      - "GET /api/fakestore/products: seller without a shop can still browse (no 403 required for shop absence)"
      - "GET /api/fakestore/products: FakeStore timeout/5xx returns 5xx with { message: string } body"
      # GET /api/fakestore/categories
      - "GET /api/fakestore/categories: public endpoint (no auth required)"
      - "GET /api/fakestore/categories: returns string array of category names from FakeStore (first call: fetches, caches)"
      - "GET /api/fakestore/categories: subsequent calls return cached data (verify via spy/mock on FakeStore client)"
      - "GET /api/fakestore/categories: on Redis unavailability, falls back to DB categories table"
      - "GET /api/fakestore/categories: on DB table empty + Redis unavailable, fetches from FakeStore and populates DB"
      - "GET /api/fakestore/categories: FakeStore timeout/5xx returns 5xx with { message: string } body"
      # POST /api/fakestore/import
      - "POST /api/fakestore/import: returns 401 without auth token"
      - "POST /api/fakestore/import: returns 403 if seller has no shop"
      - "POST /api/fakestore/import: { ids: [1, 2], overwrite: false } imports products into seller's shop"
      - "POST /api/fakestore/import: new products (no existing fakestore_id) are inserted with correct fields"
      - "POST /api/fakestore/import: overwrite=false — preserves existing price, description, stock in DB"
      - "POST /api/fakestore/import: overwrite=true — replaces price, description, stock with FakeStore values in DB"
      - "POST /api/fakestore/import: malformed FakeStore data applies field defaults (does not error)"
      - "POST /api/fakestore/import: invalid request body (missing ids, non-array, non-integer ids) returns 400"
      - "POST /api/fakestore/import: returns 200 with { importedIds: [...], count: number }"
      - "POST /api/fakestore/import: does NOT invalidate Redis categories cache"
      # No regression
      - "No regression: GET /api/products returns paginated products correctly"
      - "No regression: POST /api/products creates product correctly"
      - "No regression: PUT /api/products/:id updates product correctly"
      - "No regression: DELETE /api/products/:id deletes product correctly"

    required_services:
      - "PostgreSQL test database with users, sessions, shops, products, categories tables"
      - "Redis test instance (localhost:6379 or REDIS_URL env)"
      - "FakeStore API mocked via jest.mock() on the HTTP client module for unit tests; nock or jest.mock() for integration tests"
      - "Environment variables: DATABASE_URL, JWT_SECRET, SESSION_TTL_SECONDS, REDIS_URL, FAKESTORE_BASE_URL, FAKESTORE_TIMEOUT_MS, FAKESTORE_RETRY_ATTEMPTS, FAKESTORE_RETRY_BACKOFF_MS, FAKESTORE_CATEGORY_CACHE_TTL_SECONDS"

acceptance_criteria:
  unit: all pass
  integration: all pass
  no_regression: true

sign_off:
  coding_agent: ✅
  testing_agent: ✅
```
